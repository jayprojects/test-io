param(
    [string]$FolderPath = "Inbox",
    [string]$ReceivedAfter = "",
    [string]$ReceivedBefore = "",
    [string]$Subject = "",
    [string]$SenderEmail = "",
    [string]$IsUnread = "false",
    [string]$MaxResults = 20
)

$ol = New-Object -ComObject Outlook.Application
$namespace = $ol.GetNamespace("MAPI")

# 1. Get Folder (Defaulting to Inbox if FolderPath is blank)
$account = $namespace.GetDefaultFolder(6).Parent
try {
    $targetFolder = if ($FolderPath -ne "") { $account.Folders.Item($FolderPath) } else { $namespace.GetDefaultFolder(6) }
} catch {
    exit # Folder not found
}

# 2. Build the High-Performance Filter
$filters = @()

if ($IsUnread -eq "true") { $filters += "[UnRead] = True" }
if ($Subject -ne "")    { $filters += "[Subject] LIKE '%$Subject%'" }
if ($SenderEmail -ne "") { $filters += "[SenderEmailAddress] = '$SenderEmail'" }

if ($ReceivedAfter -ne "") {
    $afterDate = [datetime]$ReceivedAfter
    $filters += "[ReceivedTime] >= '$($afterDate.ToString("g"))'"
}
if ($ReceivedBefore -ne "") {
    $beforeDate = [datetime]$ReceivedBefore
    $filters += "[ReceivedTime] <= '$($beforeDate.ToString("g"))'"
}

# Combine filters with AND logic
$finalFilter = if ($filters.Count -gt 0) { "([ " + ($filters -join " ] AND [ ") + " ])" } else { "" }

# 3. Retrieve and Sort
$items = $targetFolder.Items
if ($finalFilter -ne "") { $items = $items.Restrict($finalFilter) }
$items.Sort("[ReceivedTime]", $true) # Sort descending (newest first)

# 4. Output results (only processing the requested MaxResults)
$count = 0
foreach ($mail in $items) {
    if ($count -ge [int]$MaxResults) { break }
    
    # Body property is only accessed for the specific 20 items we want
    $body = $mail.Body -replace "[\r\n]", " "
    Write-Output "$($mail.Subject)|||$($mail.SenderName)|||$($mail.ReceivedTime)|||$body"
    
    $count++
}
