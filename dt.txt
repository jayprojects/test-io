package hello;
//System.setProperty("webdriver.chrome.driver", "C:\\PortableApps\\chromedriver-win64-146\\chromedriver.exe");
import org.openqa.selenium.*;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.*;
import java.io.*;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.List;

public class CSEAnnouncementsScraper {

    public static void main(String[] args) {
        // Set up WebDriver
        System.setProperty("webdriver.chrome.driver", "C:\\PortableApps\\chromedriver-win64-146\\chromedriver.exe");
        WebDriver driver = new ChromeDriver();

        try {
            driver.get("https://www.cse.lk/announcements");

            // Wait for the page to load completely
            WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(20));
            wait.until(ExpectedConditions.presenceOfElementLocated(By.tagName("h3")));  // Waiting for h3 elements to load (table titles)

            // Table titles with respective CSV file names
            String[] tableTitles = {
                    "DIVIDEND ANNOUNCEMENTS",
                    "LISTED COMPANIES – NOTICE OF ANNUAL GENERAL MEETINGS (AGM)",
                    "LISTED COMPANIES – NOTICE OF EXTRAORDINARY GENERAL MEETINGS (EGM)"
            };

            String[] outputFiles = {
                    "Cash_Dividend_LK.csv",
                    "Annual_General_Meeting_LK.csv",
                    "Extraordinary_General_meeting_LK.csv"
            };

            for (int i = 0; i < tableTitles.length; i++) {
                String title = tableTitles[i];
                String outputFileName = outputFiles[i];

                try {
                    WebElement tableTitleElement = driver.findElement(By.xpath("//h3[contains(text(), '" + title + "')]"));
                    if (tableTitleElement != null) {
                        System.out.println("Found table for: " + title);

                        // Locate the parent div for the table
                        WebElement tableParentDiv = tableTitleElement.findElement(By.xpath("./ancestor::div[@class='mb-8']"));
                        WebElement table = tableParentDiv.findElement(By.xpath(".//table[contains(@class, 'app-table')]"));

                        if (table != null) {
                            // Extract table data with correct formatting
                            String csvData = extractTableData(table);
                            writeToCSV(csvData, outputFileName);
                            System.out.println("CSV file created for: " + title);
                        }
                    }
                } catch (NoSuchElementException e) {
                    System.out.println("No table found for: " + title);
                    writeToCSV("", outputFileName);  // Create empty CSV if table is not found
                }
            }

        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            driver.quit();
        }
    }

    // Extract the data from the table and format it as a comma-separated CSV with double quotes around each field
    private static String extractTableData(WebElement table) {
        StringBuilder data = new StringBuilder();

        // Get the headers dynamically from the table
        List<WebElement> headers = table.findElements(By.xpath(".//thead//th"));
        String[] headerColumns = new String[headers.size()];

        for (int i = 0; i < headers.size(); i++) {
            headerColumns[i] = "\"" + headers.get(i).getText().trim().replace("\"", "\"\"") + "\""; // Enclose headers in quotes
        }
        // Join headers with commas and add to the CSV data
        data.append(String.join(",", headerColumns) + "\n");

        // Get all rows from the table
        List<WebElement> rows = table.findElements(By.tagName("tr"));

        // Loop through each row
        for (WebElement row : rows) {
            List<WebElement> columns = row.findElements(By.tagName("td"));

            if (columns.size() > 0) {  // Only process rows with data (ignores empty rows)
                String[] rowData = new String[columns.size()];

                for (int i = 0; i < columns.size(); i++) {
                    rowData[i] = "\"" + columns.get(i).getText().trim().replace("\"", "\"\"") + "\"";  // Enclose each cell in quotes
                }

                // Join row data with commas and add to the CSV data
                data.append(String.join(",", rowData) + "\n");
            }
        }

        return data.toString();
    }

    // Write data to a CSV file at C:\temp with UTF-8 encoding
    private static void writeToCSV(String data, String fileName) {
        File directory = new File("C:\\temp");
        if (!directory.exists()) {
            directory.mkdirs(); // Ensure the directory exists
        }

        String filePath = "C:\\temp\\" + fileName;

        // Use OutputStreamWriter with UTF-8 encoding explicitly
        try (BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(new FileOutputStream(filePath), StandardCharsets.UTF_8))) {
            writer.write(data);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
