@Test
public void testWithMockConstruction() {
    try (MockedConstruction<ProcessBuilder> mocked = mockConstruction(ProcessBuilder.class, 
        (mock, context) -> {
            // Configure the mock behavior
            Process mockProcess = mock(Process.class);
            when(mock.start()).thenReturn(mockProcess);
            // Setup input streams if your code reads output
            when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("done".getBytes()));
        })) {

        ScriptRunner runner = new ScriptRunner();
        runner.executeLogic();

        // The code inside executeLogic will now get a mock ProcessBuilder
        assertEquals(1, mocked.constructed().size());
    }
}
