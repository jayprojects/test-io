using System;
using System.Diagnostics;
using System.Windows.Forms;

namespace GitHubToClipboard
{
    class Program
    {
        [STAThread]
        static void Main(string[] args)
        {
            string rawUrl = "https://raw.githubusercontent.com/jayprojects/test-io/main/dt.txt";
            
            // --- SCENARIO A: Public Repository ---
            // string arguments = $"-s -L \"{rawUrl}\"";

            // --- SCENARIO B: Private Repository (You likely need this) ---
            // Replace YOUR_GITHUB_PAT with your actual classic token or fine-grained token
            string arguments = $"-s -L -H \"Authorization: token YOUR_GITHUB_PAT\" \"{rawUrl}\"";

            try
            {
                Console.WriteLine("Fetching text via curl...");

                ProcessStartInfo psi = new ProcessStartInfo
                {
                    FileName = "curl",          // Calls the native Windows curl.exe
                    Arguments = arguments,      // Passes the flags and URL
                    RedirectStandardOutput = true,
                    RedirectStandardError = true,
                    UseShellExecute = false,    // Required to read the output
                    CreateNoWindow = true       // Hides the command prompt
                };

                using (Process process = Process.Start(psi))
                {
                    // Read the downloaded text from standard output
                    string output = process.StandardOutput.ReadToEnd();
                    string error = process.StandardError.ReadToEnd();
                    process.WaitForExit();

                    if (process.ExitCode == 0)
                    {
                        // GitHub raw URLs return plain text errors like "404: Not Found" if authorization fails
                        if (output.Contains("404: Not Found") || output.Contains("403: Forbidden") || output.Trim() == "404: Not Found")
                        {
                            Console.WriteLine($"GitHub denied access. Check your token/URL. \nResponse: {output.Trim()}");
                            return;
                        }

                        if (!string.IsNullOrEmpty(output))
                        {
                            Clipboard.SetText(output);
                            Console.WriteLine("Success! The text has been copied to your clipboard.");
                        }
                        else
                        {
                            Console.WriteLine("The file was empty.");
                        }
                    }
                    else
                    {
                        Console.WriteLine($"Curl failed with Exit Code: {process.ExitCode}\nError Stream: {error}");
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"An error occurred trying to run curl: {ex.Message}");
            }
        }
    }
}
