function Find-CertificateStore {
    param (
        [Parameter(Mandatory=$true, HelpMessage="Enter the certificate thumbprint")]
        [string]$Thumbprint
    )

    # Clean the thumbprint: MMC famously copies hidden characters and spaces, 
    # so this strips everything except valid hex characters.
    $cleanThumbprint = $Thumbprint -replace '[^a-fA-F0-9]', ''

    Write-Host "Searching for Thumbprint: $cleanThumbprint..." -ForegroundColor Cyan

    # Search all certificate stores recursively
    $foundCerts = Get-ChildItem -Path Cert:\ -Recurse | Where-Object { $_.Thumbprint -eq $cleanThumbprint }

    if (-not $foundCerts) {
        Write-Warning "Could not find a certificate with thumbprint '$cleanThumbprint' in any keystore."
        return
    }

    # Process and format the output for each match found
    $results = foreach ($cert in $foundCerts) {
        # Strip the messy PowerShell provider prefix to get just Location\Store
        $systemPath = $cert.PSParentPath -replace ".*::", ""
        
        # Split the path to get just the folder name (e.g., 'My', 'Root')
        $storeName = $systemPath.Split('\')[-1]

        # Translate PowerShell internal names to the MMC UI friendly names
        $friendlyStoreName = switch ($storeName) {
            'My'               { 'Personal' }
            'Root'             { 'Trusted Root Certification Authorities' }
            'CA'               { 'Intermediate Certification Authorities' }
            'AuthRoot'         { 'Third-Party Root Certification Authorities' }
            'TrustedPublisher' { 'Trusted Publishers' }
            'Disallowed'       { 'Untrusted Certificates' }
            'ClientAuthIssuer' { 'Client Authentication Issuers' }
            default            { $storeName } 
        }

        # Swap the system folder name for the friendly folder name in the path string
        $friendlyPath = $systemPath -replace "\\$storeName$", "\$friendlyStoreName"

        [PSCustomObject]@{
            Subject        = $cert.Subject
            SystemPath     = "Cert:\$systemPath"
            MMC_UIDisplay  = $friendlyPath
        }
    }

    # Display the results neatly
    $results | Format-List
}
