param(
    [string]$Subject = "",
    [string]$AttachmentName = "",
    [string]$DestinationPath = "",
    [string]$IsUnread = "false",
    [string]$MarkRead = "false",
    [string]$FolderPath = "Inbox"
)

$ol = New-Object -ComObject Outlook.Application
$namespace = $ol.GetNamespace("MAPI")

# 1. Direct Folder Access
$account = $namespace.GetDefaultFolder(6).Parent
try {
    $targetFolder = if ($FolderPath -ne "") { $account.Folders.Item($FolderPath) } else { $namespace.GetDefaultFolder(6) }
} catch {
    Write-Host "ERROR: Folder not found"
    exit
}

# 2. Build Filter
$filters = @()
if ($IsUnread -eq "true") { $filters += "[UnRead] = True" }
if ($Subject -ne "")     { $filters += "[Subject] LIKE '%$Subject%'" }
# High-performance check: only look for emails that actually have attachments
$filters += "[HasAttachment] = True"

$finalFilter = "([ " + ($filters -join " ] AND [ ") + " ])"

# 3. Find the most recent match
$items = $targetFolder.Items
$items.Sort("[ReceivedTime]", $true) # Sort descending so the first match is the newest
$matchedMail = $items.Find($finalFilter)

if ($null -eq $matchedMail) {
    Write-Host "ERROR: No email found matching this criteria"
    exit
}

# 4. Process Attachment
$found = $false
foreach ($attachment in $matchedMail.Attachments) {
    if ($attachment.FileName -eq $AttachmentName) {
        $fullPath = Join-Path $DestinationPath $AttachmentName
        $attachment.SaveAsFile($fullPath)
        $found = $true
        break
    }
}

# 5. Handle Post-Download Actions
if ($found) {
    if ($MarkRead -eq "true") {
        $matchedMail.UnRead = $false
        $matchedMail.Save()
    }
    Write-Host "SUCCESS"
} else {
    Write-Host "ERROR: Attachment not found"
}
