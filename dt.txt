using System;
using System.Net; // Required for WebRequest and CredentialCache
using System.Net.Http;
using System.Windows.Forms;

namespace GitHubToClipboard
{
    class Program
    {
        [STAThread]
        static void Main(string[] args)
        {
            string rawUrl = "https://raw.githubusercontent.com/jayprojects/test-io/main/dt.txt";

            try
            {
                Console.WriteLine("Fetching text from GitHub through proxy...");

                // 1. Configure the HttpClientHandler for the proxy
                HttpClientHandler handler = new HttpClientHandler
                {
                    UseProxy = true,
                    // Grab the default proxy settings from Windows/IE
                    Proxy = WebRequest.GetSystemWebProxy(), 
                    // Tell the HttpClient to use your Windows login for the request
                    UseDefaultCredentials = true 
                };

                // 2. Explicitly pass your Windows credentials to the Proxy specifically
                if (handler.Proxy != null)
                {
                    handler.Proxy.Credentials = CredentialCache.DefaultCredentials;
                }

                // 3. Inject the handler into the HttpClient
                using (HttpClient client = new HttpClient(handler))
                {
                    // GitHub STILL requires a User-Agent, so we keep this.
                    client.DefaultRequestHeaders.Add("User-Agent", "MyCSharpApp");

                    // Fetch the text
                    string textData = client.GetStringAsync(rawUrl).GetAwaiter().GetResult();

                    if (!string.IsNullOrEmpty(textData))
                    {
                        Clipboard.SetText(textData);
                        Console.WriteLine("Success! The text has been copied to your clipboard.");
                    }
                    else
                    {
                        Console.WriteLine("The file was empty.");
                    }
                }
            }
            catch (HttpRequestException httpEx)
            {
                // If it fails now, it will usually give a more descriptive proxy error here
                Console.WriteLine($"Network or Proxy Error: {httpEx.Message}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"An error occurred: {ex.Message}");
            }
        }
    }
}
